cmake_minimum_required(VERSION 3.25)
project(NcmaDependencyTrackingFixture LANGUAGES C CXX)
if(NOT MSVC OR NOT CMAKE_GENERATOR STREQUAL "Ninja")
    message(FATAL_ERROR "This regression exercises actual MSVC/Ninja dependency parsing")
endif()
if(NOT DEFINED TEST_PREFIX_FILE OR NOT DEFINED TRACKING_MODULE)
    message(FATAL_ERROR "Missing parser fixture inputs")
endif()
include("${TEST_PREFIX_FILE}")
set(CMAKE_CL_SHOWINCLUDES_PREFIX "${TEST_PREFIX}")
include("${TRACKING_MODULE}")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(ProbeC OBJECT probe.c)
add_library(ProbeCxx OBJECT probe.cpp)
target_compile_options(ProbeC PRIVATE /W4 /WX /utf-8)
target_compile_options(ProbeCxx PRIVATE /W4 /WX /utf-8)
